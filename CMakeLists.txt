set(CMAKE_WARN_DEPRECATED OFF CACHE INTERNAL "")

cmake_minimum_required(VERSION 3.21)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake")
include(GlobalOptions)
add_global_options(CXX_STANDARD 17)
include(CustomCompilerOptions)
fix_compile_flags()
if(EMSCRIPTEN)
    fix_release_flags(STRIP_BINARY OFF USE_LTO ON USE_STATIC_CRT OFF)
else()
    fix_release_flags(STRIP_BINARY ON USE_LTO ON USE_STATIC_CRT ON)
endif()
include(ProjectMacros)

option(BUILD_VIEWER "Build viewer application" OFF)
option(DSP_BUILD_ORACLE "Build the bit-exact parity oracle" OFF)
option(DSP_BUILD_TESTS "Build deterministic generator unit tests" OFF)
option(DSP_BUILD_BENCHMARK_PLUGINS "Build synthetic benchmark plugins" OFF)
option(DSP_ENABLE_INSTRUMENTATION "Enable read-only performance counters" OFF)
option(DSP_ENABLE_NATIVE_ARCH "Enable native CPU tuning for local benchmarks" OFF)
set(DSP_PGO "OFF" CACHE STRING "PGO mode: OFF, GENERATE, or USE")
set_property(CACHE DSP_PGO PROPERTY STRINGS OFF GENERATE USE)
set(DSP_PGO_PROFILE_DIR "" CACHE PATH "Directory containing compiler PGO data")
set(DSP_PGO_PROFILE_DATA "" CACHE FILEPATH "Merged Clang PGO profile")

if(NOT DSP_PGO STREQUAL "OFF" AND NOT DSP_PGO STREQUAL "GENERATE" AND NOT DSP_PGO STREQUAL "USE")
    message(FATAL_ERROR "DSP_PGO must be OFF, GENERATE, or USE")
endif()
project(DSPSeedCalc CXX)

add_subdirectory(dspugen)

if(WIN32)
    add_subdirectory(getopt)
    add_subdirectory(dlfcn-win32)
endif()

add_subdirectory(cli)

set(DSP_PGO_TARGETS ${PROJECT_NAME} dspugen)
if(DSP_ENABLE_NATIVE_ARCH)
    if(MSVC)
        message(FATAL_ERROR "DSP_ENABLE_NATIVE_ARCH is not implemented for MSVC")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        foreach(target IN LISTS DSP_PGO_TARGETS)
            target_compile_options(${target} PRIVATE -march=native)
        endforeach()
    endif()
endif()

if(DSP_PGO STREQUAL "GENERATE")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        if(NOT DSP_PGO_PROFILE_DIR)
            message(FATAL_ERROR "DSP_PGO_PROFILE_DIR is required for GCC DSP_PGO=GENERATE")
        endif()
        foreach(target IN LISTS DSP_PGO_TARGETS)
            target_compile_options(${target} PRIVATE -fprofile-generate=${DSP_PGO_PROFILE_DIR} -fprofile-update=atomic)
            target_link_options(${target} PRIVATE -fprofile-generate=${DSP_PGO_PROFILE_DIR} -fprofile-update=atomic)
        endforeach()
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        foreach(target IN LISTS DSP_PGO_TARGETS)
            target_compile_options(${target} PRIVATE -fprofile-instr-generate)
            target_link_options(${target} PRIVATE -fprofile-instr-generate)
        endforeach()
    else()
        message(FATAL_ERROR "DSP_PGO=GENERATE supports GCC and Clang only")
    endif()
elseif(DSP_PGO STREQUAL "USE")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        if(NOT DSP_PGO_PROFILE_DIR)
            message(FATAL_ERROR "DSP_PGO_PROFILE_DIR is required for GCC DSP_PGO=USE")
        endif()
        foreach(target IN LISTS DSP_PGO_TARGETS)
            target_compile_options(${target} PRIVATE -fprofile-use=${DSP_PGO_PROFILE_DIR} -fprofile-correction -Wmissing-profile)
            target_link_options(${target} PRIVATE -fprofile-use=${DSP_PGO_PROFILE_DIR} -fprofile-correction)
        endforeach()
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        if(NOT DSP_PGO_PROFILE_DATA)
            message(FATAL_ERROR "DSP_PGO_PROFILE_DATA is required for Clang DSP_PGO=USE")
        endif()
        foreach(target IN LISTS DSP_PGO_TARGETS)
            target_compile_options(${target} PRIVATE -fprofile-instr-use=${DSP_PGO_PROFILE_DATA})
            target_link_options(${target} PRIVATE -fprofile-instr-use=${DSP_PGO_PROFILE_DATA})
        endforeach()
    else()
        message(FATAL_ERROR "DSP_PGO=USE supports GCC and Clang only")
    endif()
endif()
add_subdirectory(filters)

if(DSP_BUILD_ORACLE)
    add_executable(dspugen_oracle tests/oracle.cc)
    target_link_libraries(dspugen_oracle PRIVATE dspugen)
    target_compile_features(dspugen_oracle PRIVATE cxx_std_17)
    if(DSP_PGO STREQUAL "GENERATE")
        if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
            target_link_options(dspugen_oracle PRIVATE -fprofile-generate=${DSP_PGO_PROFILE_DIR} -fprofile-update=atomic)
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            target_link_options(dspugen_oracle PRIVATE -fprofile-instr-generate)
        endif()
    elseif(DSP_PGO STREQUAL "USE")
        if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
            target_link_options(dspugen_oracle PRIVATE -fprofile-use=${DSP_PGO_PROFILE_DIR} -fprofile-correction)
        elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
            target_link_options(dspugen_oracle PRIVATE -fprofile-instr-use=${DSP_PGO_PROFILE_DATA})
        endif()
    endif()
    set_target_properties(dspugen_oracle PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
endif()

    set_target_properties(dspugen_vein_tests PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
